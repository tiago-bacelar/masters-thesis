type Identifier = String

--TODO: trig functions?
data LTerm a = Scalar a | Prod a Identifier deriving (Show)
type LExp a = [LTerm a]

data BTerm a = T
             | F
             | LT  (LExp a) (LExp a)
             | GT  (LExp a) (LExp a)
             | LEQ (LExp a) (LExp a)
             | GEQ (LExp a) (LExp a)
             | EQ  (LExp a) (LExp a)
             | NEQ (LExp a) (LExp a) deriving (Show)
--TODO: BExp (and, or, not)

data Statement a = Assign Identifier (LExp a)
                 | For [(Identifier, LExp a)] (LExp a)
                 | IfThenElse (BTerm a) (Program a) (Program a)
                 | WhileDo (BTerm a) (Program a) deriving (Show)
type Program a = [Statement a]

{-

program: (statement ';')+
statement: atomic | 'if' bterm 'then' program ('else' program)? | 'while' bterm 'do' program
atomic: VAR ':=' lexp | difeq (',' difeq)* 'for' lexp | 'wait' lexp
difeq: VAR' '=' lexp

def = emptyDef{ commentLine = "//"
              , commentStart = "/*"
              , commentEnd = "*/"
              , identStart = letter
              , identLetter = alphaNum
              , reservedNames = ["true", "false", "if", "then", "else", "while", "do", "for", "wait"]
              }

TokenParser{ identifier = m_identifier
           , symbol = m_symbol
           , reserved = m_reserved
           , commaSep1 = m_commaSep1
           , whiteSpace = m_whiteSpace } = makeTokenParser def

num :: (Fractional a) => Parser a
num = (fromRational <$>) $ (toRational <$> integer) <+=> frac <*=> exponent <* m_whiteSpace
       where number = (read :: String -> Integer) <$> many1 digit
             plus = char '+' >> number
             minus = char '-' >> negate <$> number
             integer = plus <|> minus <|> number
             base = 10 :: Integer
             frac  = option 0 $ char '.' >> (\s -> read s % base ^ length s) <$> many1 digit
             exponent = option 1 $ oneOf "eE" >> (toRational base ^^) <$> integer
             (<+=>) = (<*>) . ((+) <$>)
             (<*=>) = (<*>) . ((*) <$>)


comp :: Parser (LExp a -> LExp a -> BTerm a)
comp = choice [m_symbol a $> b | (a,b) <- [("<", LT), (">", GT), ("<=", LEQ), (">=", GEQ), ("==", EQ), ("!=", NEQ)]]

bterm :: (Fractional a) => Parser (BTerm a)
bterm = (m_reserved "true" $> T)
        <|> (m_reserved "false" $> F)
        <|> do { l1 <- lexp; c <- comp; l2 <- lexp; return (c l1 l2) }

lterm :: (Fractional a) => Parser (LTerm a)
lterm = choice [
              try $ do { x <- num; m_symbol "*"; y <- num; return . Scalar $ x*y },
              try $ do { x <- num; m_symbol "/"; y <- num; return . Scalar $ x/y },
              try $ do { x <- num; m_symbol "*"; y <- m_identifier; return $ Prod x y },
              try $ do { x <- m_identifier; m_symbol "*"; y <- num; return $ Prod y x },
              try $ do { x <- m_identifier; m_symbol "/"; y <- num; return $ Prod (recip y) x },
              Scalar <$> num,
              Prod 1 <$> m_identifier
                ]

lexp :: (Fractional a) => Parser (LExp a)
lexp = chainl1 (singleton <$> lterm) lop --this is a bit ugly because we need to use lists for chainl1. maybe refactor later
       where sim (Scalar s) = Scalar (-s)
             sim (Prod s v) = Prod (-s) v
             lop = m_symbol "+" $> (++)
                   <|> m_symbol "-" $> (\x y -> x ++ (map sim y))


ifThenElse, whileDo, wait, assign, for :: (Fractional a) => Parser (Statement a)
ifThenElse = do { m_reserved "if"; b <- bterm; m_reserved "then"; p <- program; q <- option [] (m_reserved "else" >> program); return (IfThenElse b p q) }
whileDo = do { m_reserved "while"; b <- bterm; m_reserved "do"; p <- program; return (WhileDo b p) }
wait = For [] <$> (m_reserved "wait" >> lexp)
assign = do { v <- m_identifier; m_symbol ":="; l <- lexp; return (Assign v l)}
for = do { r <- m_commaSep1 difeq; m_reserved "for"; l <- lexp; return (For r l) } --TODO: check for unassigned variables
       where difeq = do { v <- m_identifier; m_symbol "'"; m_symbol "="; l <- lexp; return (v,l) }

statement :: (Fractional a) => Parser (Statement a)
statement = ifThenElse <|> whileDo <|> wait <|> try assign <|> for

--TODO: parenthesis in programs (dangling else)
program :: (Fractional a) => Parser (Program a)
program = statement `sepEndBy1` (m_symbol ";")

mainparser :: (Fractional a) => Parser (Program a)
mainparser = m_whiteSpace >> program <* eof

parseJaguar :: (Fractional a) => String -> Either ParseError (Program a)
parseJaguar = parse mainparser ""


playP :: IO ()
playP = do
       input <- readFile "input.txt"
       case parseJaguar input of
              Left err -> print err
              Right ans -> print (ans :: Program Rational)
